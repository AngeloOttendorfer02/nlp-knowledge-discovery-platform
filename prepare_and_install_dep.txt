# NLP Knowledge Discovery Platform
# Reproducible Setup, Installation, and Validation Guide
#
# Recommended platform:
# - Linux
# - Conda/Miniforge
# - Python 3.11
#
# Important:
# Run all commands from the repository root unless a command explicitly says
# otherwise. Do not use a personal absolute path. After cloning, enter the
# repository directory and resolve the root with Git as shown below.

###############################################################################
# 1. Clone the repository and enter the project root
###############################################################################

# Replace <repository-url> with the Git repository URL.
git clone <repository-url>
cd nlp-knowledge-discovery-platform

# Optional verification: this must print the repository root.
git rev-parse --show-toplevel


###############################################################################
# 2. Install Miniforge only if Conda is not available
###############################################################################

conda --version

# If the command above fails with "conda: command not found", install Miniforge:
cd "${HOME}"
wget -O Miniforge3.sh \
  https://github.com/conda-forge/miniforge/releases/latest/download/Miniforge3-Linux-x86_64.sh
bash Miniforge3.sh

# Reopen the terminal after installation, or initialize Conda in the current
# shell when Miniforge was installed in the default location:
source "${HOME}/miniforge3/etc/profile.d/conda.sh"
conda --version

# Return to the cloned repository before continuing.
cd /path/to/nlp-knowledge-discovery-platform


###############################################################################
# 3. Create and activate the project environment
###############################################################################

conda create -n nlp-kg python=3.11 -y
conda activate nlp-kg

python --version
which python

# Expected:
# - Python 3.11.x
# - the Python executable is inside the nlp-kg Conda environment


###############################################################################
# 4. Install project dependencies
###############################################################################

python -m pip install --upgrade pip setuptools wheel
python -m pip install -r requirements.txt

# Kaggle CLI is used by the full validation script to download the public
# arXiv metadata dataset when it is not already available locally.
python -m pip install kaggle

# Install the English spaCy model used by the extraction pipeline.
python -m spacy download en_core_web_sm

# Verify the installed dependency set.
python -m pip check


###############################################################################
# 5. Optional Jupyter kernel registration
###############################################################################

# Register the environment so it can be selected in JupyterLab or Notebook.
python -m ipykernel install \
  --user \
  --name nlp-kg \
  --display-name "Python (nlp-kg)"


###############################################################################
# 6. Recommended one-command end-to-end validation
###############################################################################

chmod +x scripts/run_full_validation.sh

# Complete validation including the Streamlit dashboard:
./scripts/run_full_validation.sh

# Complete validation without starting the dashboard:
./scripts/run_full_validation.sh --no-dashboard

# Reuse an already prepared environment and skip package installation:
./scripts/run_full_validation.sh --skip-install --no-dashboard

# Show all available options:
./scripts/run_full_validation.sh --help

# The script performs the complete reproducible workflow:
# - validates the active Conda environment and Python interpreter;
# - installs or validates dependencies;
# - validates the spaCy model;
# - runs compileall and pytest;
# - downloads the public Cornell arXiv metadata dataset when it is missing;
# - creates a reproducible real-world subset;
# - runs preprocessing, extraction, topic modeling, and graph construction;
# - evaluates BM25, semantic, and KG-enhanced retrieval;
# - generates report tables and figures from real metrics;
# - runs the manually labelled qualitative retrieval case study;
# - writes terminal output to logs/e2e/;
# - optionally starts the Streamlit dashboard.

# Notes about the dataset download:
# - the full metadata file is approximately 5 GB;
# - the public dataset normally downloads without an API token;
# - Kaggle may still request authentication in some environments or after
#   service-side policy or rate-limit changes;
# - the file is not downloaded again when it already exists at:
#   data/raw/arxiv-metadata-oai-snapshot.json

###############################################################################
# 12. Troubleshooting
###############################################################################

# Problem: the Python interpreter is outside the Conda environment.
# Solution:
conda activate nlp-kg
which python
python --version

# Problem: sentence_transformers or networkx cannot be imported.
# Solution:
conda activate nlp-kg
python -m pip install -r requirements.txt

# Problem: spaCy cannot load en_core_web_sm.
# Solution:
python -m spacy download en_core_web_sm

# Problem: Kaggle download is rejected.
# Solution:
# The dataset is public and normally downloads anonymously. If Kaggle requests
# authentication in the current environment, configure the Kaggle CLI according
# to its current login instructions and rerun the validation script.

# Problem: the full validation script stops.
# Solution:
tail -n 150 logs/e2e/latest.log

# Problem: generated files from an old run affect the result.
# Solution:
# The default full validation removes old generated outputs. To preserve old
# outputs, use --keep-outputs explicitly.
